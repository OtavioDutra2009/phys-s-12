int LED = A3;
int shortTime = 300;
int longTime = 900;
int space = 1800;

void setup() {
  pinMode(LED, OUTPUT);
}

void loop() {
  //O
  switchLED(longTime);
  switchLED(longTime);
  switchLED(longTime);
  delay(longTime);

  //T
  switchLED(longTime);
  delay(longTime);

  //A
  switchLED(shortTime);
  switchLED(longTime);
  delay(longTime);

  //V
  switchLED(shortTime);
  switchLED(shortTime);
  switchLED(shortTime);
  switchLED(longTime);
  delay(longTime);

  //I
  switchLED(shortTime);
  switchLED(shortTime);
  delay(longTime);

  //O
  switchLED(longTime);
  switchLED(longTime);
  switchLED(longTime);
  delay(longTime);

  //SPACE
  delay(space);

  //D
  switchLED(longTime);
  switchLED(shortTime);
  switchLED(shortTime);
  delay(longTime);

  //U
  switchLED(shortTime);
  switchLED(shortTime);
  switchLED(longTime);
  delay(longTime);

  //T
  switchLED(longTime);
  delay(longTime);

  //R
    switchLED(shortTime);
    switchLED(longTime);
    switchLED(shortTime);
    delay(longTime);

    //A
  switchLED(shortTime);
  switchLED(longTime);
  delay(longTime);

  //space
  delay(space);
}

void switchLED(int timing) {
  digitalWrite(LED, HIGH);
  delay(timing);
  digitalWrite(LED, LOW);
  delay(shortTime);
}
