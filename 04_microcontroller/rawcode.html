int shortTime = 300;

  // put your setup code here, to run once:
  pinMode(LED, OUTPUT);  // make pin D3 output
}

  // put your main code here, to run repeatedly:
  //O
  switchLED(longTime);

  switchLED(longTime);
  delay(longTime);

  //T
  switchLED(longTime);

  //A

  switchLED(shortTime);

  //O

void switchLED(int timing) {
  digitalWrite(LED, HIGH);  // send some voltage to pin A3
  delay(timing);            // short delay
  digitalWrite(LED, LOW);
  delay(shortTime);  // short delay
}
