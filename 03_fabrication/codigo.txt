float calculatePercent(float grams) {
  float percent =
    ((grams - EMPTY_SHELL_GRAMS) /
    (FULL_SHELL_GRAMS - EMPTY_SHELL_GRAMS)) * 100.0;

  return constrain(percent, 0.0, 100.0);
}

void handleRoot() {
  server.send(200, "text/html", PAGE);
}

void handleWeight() {
  if (!scale.wait_ready_timeout(1000)) {
    server.send(
      503,
      "application/json",
      "{\"error\":\"HX711 not ready\"}"
    );
    return;
  }

  float grams = scale.get_units(20);

  if (grams <= 0.5) {
    grams = 0;
  }

  float percent = calculatePercent(grams);
  float simulatedKg =
    (percent / 100.0) * LPG_CAPACITY_KG;

  String json = "{\"grams\":";
  json += String(grams, 2);
  json += ",\"percent\":";
  json += String(percent, 2);
  json += ",\"simulatedKg\":";
  json += String(simulatedKg, 2);
  json += "}";

  server.send(200, "application/json", json);

  Serial.print("Measured: ");
  Serial.print(grams, 1);
  Serial.print(" g | LPG simulation: ");
  Serial.print(percent, 1);
  Serial.print("% | ");
  Serial.print(simulatedKg, 2);
  Serial.println(" kg");
}

void handleTare() {
  if (!scale.wait_ready_timeout(1000)) {
    server.send(
      503,
      "application/json",
      "{\"error\":\"HX711 not ready\"}"
    );
    return;
  }

  scale.tare(60);

  server.send(
    200,
    "application/json",
    "{\"ok\":true}"
  );

  Serial.println("Current load zeroed.");
}

void handleBaseZero() {
  if (!scale.wait_ready_timeout(1000)) {
    server.send(
      503,
      "application/json",
      "{\"error\":\"HX711 not ready\"}"
    );
    return;
  }

  scale.set_offset(baseOffset);

  server.send(
    200,
    "application/json",
    "{\"ok\":true}"
  );

  Serial.println("Original base zero restored.");
}

void setup() {
  Serial.begin(115200);

  scale.begin(HX711_DOUT, HX711_SCK);
  scale.set_scale(CALIBRATION_FACTOR);

  Serial.println("Keep the base empty.");
  Serial.println("Letting the load cell settle...");

  delay(5000);

  scale.wait_ready_timeout(3000);
  scale.tare(60);

  baseOffset = scale.get_offset();

  Serial.println("Base tared.");

  Serial.print("Connecting to ");
  Serial.println(ssid);

  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi connected.");

  Serial.print("Open this address: http://");
  Serial.println(WiFi.localIP());

  server.on("/", HTTP_GET, handleRoot);
  server.on("/api/weight", HTTP_GET, handleWeight);
  server.on("/api/tare", HTTP_POST, handleTare);
  server.on("/api/base-zero", HTTP_POST, handleBaseZero);

  server.begin();

  Serial.println("Original LPG simulation ready.");
}

void loop() {
  server.handleClient();
}
